Requirement 7: Home Folders with Permissions and Quota

🎯 QUOTA

100 MB
# HomeFolders-FULL-AUTOMATION.ps1
# Run on FILE SERVER as Administrator

Import-Module ServerManager
$ServerName = $env:COMPUTERNAME
$HomePath = "D:\HomeFolders"
$Classes = @("CT1A","CT1B","CT2A","CT2B","CT3A","CT3B","CT4A","CT4B")

Write-Host "🚀 STEP 1: Create Folder Structure" -ForegroundColor Cyan
New-Item -Path $HomePath -ItemType Directory -Force
foreach ($class in $Classes) { 
    New-Item -Path "$HomePath\$class" -ItemType Directory -Force 
    Write-Host "  📁 $HomePath\$class" -ForegroundColor Green
}

Write-Host "`n🚀 STEP 2: FSRM Role + Share" -ForegroundColor Cyan
# Install FSRM if missing
if (!(Get-WindowsFeature FS-Resource-Manager).Installed) {
    Install-WindowsFeature -Name FS-Resource-Manager -IncludeManagementTools
}

# Create hidden share
New-SmbShare -Name "HomeFolders$" -Path $HomePath -FullAccess "Administrators" -ChangeAccess "Authenticated Users" -Force

Write-Host "`n🚀 STEP 3: Bulk Student Folders + NTFS" -ForegroundColor Cyan
Import-Module ActiveDirectory
foreach ($class in $Classes) {
    $students = Get-ADUser -Filter * -SearchBase "OU=Users,OU=$class,DC=group1,DC=local"
    $teacher = Get-ADUser -Filter "Title -like '*Teacher*'" -SearchBase "OU=Teachers,OU=$class,DC=group1,DC=local" | Select -First 1
    
    foreach ($student in $students) {
        $folder = "$HomePath\$class\$($student.SamAccountName)-$class"
        New-Item -Path $folder -ItemType Directory -Force
        
        # NTFS PERMISSIONS (Your exact spec)
        icacls $folder /reset /T
        icacls $folder /inheritance:d
        icacls $folder /grant "${student.SamAccountName}:(OI)(CI)M"        # Student: Modify
        icacls $folder /grant "${teacher.SamAccountName}:(OI)(CI)F"        # Teacher: Full  
        icacls $folder /grant "Administrators:(OI)(CI)F"                   # Admins: Full
        icacls $folder /grant "CREATOR OWNER:(OI)(CI)(IO)(MU)" /T          # Creator Owner
        
        Write-Host "  👤 $($student.SamAccountName) → $folder" -ForegroundColor Yellow
    }
}

Write-Host "`n🚀 STEP 4: FSRM 100MB Quota (Per Student)" -ForegroundColor Cyan
foreach ($class in $Classes) {
    $students = Get-ADUser -Filter * -SearchBase "OU=Users,OU=$class,DC=group1,DC=local"
    foreach ($student in $students) {
        $quotaPath = "$HomePath\$class\$($student.SamAccountName)-$class"
        if (Test-Path $quotaPath) {
            dirquota quota add $quotaPath "100MB" "HardQuota" "C:\Windows\QuotaTemplate.xml"
        }
    }
}

Write-Host "`n🚀 STEP 5: AD Home Folder Mapping" -ForegroundColor Cyan
foreach ($class in $Classes) {
    $students = Get-ADUser -Filter * -SearchBase "OU=Users,OU=$class,DC=group1,DC=local"
    foreach ($student in $students) {
        Set-ADUser $student -HomeDirectory "\\$ServerName\HomeFolders$\$class\$($student.SamAccountName)-$class" -HomeDrive "H:"
    }
}

Write-Host "`n✅ COMPLETE! Verify:" -ForegroundColor Green
Write-Host "   dirquota quota list" -ForegroundColor White
Write-Host "   echo %HOMEDRIVE% (student login)" -ForegroundColor White
