Requirement 5: Logon Time Restrictions per User

🚀 Complete PowerShell Script:

Import-Module ActiveDirectory function New-LogonArray { return New-Object byte[] 21 } function Set-Hour { param($array, $day, $start, $end) for ($h = $start; $h -lt $end; $h++) { $index = ($day * 24) + $h $byte = [math]::Floor($index / 8) $bit = $index % 8 $array[$byte] = $array[$byte] -bor (1 -shl $bit) } } # CT1A: Mon-Fri 00:00-09:00 $ct1a = New-LogonArray for ($d=1; $d -le 5; $d++) { Set-Hour $ct1a $d 0 9 } Get-ADUser -Filter * -SearchBase "OU=CT1A,DC=group1,DC=local" | ForEach-Object { Set-ADUser $_ -Replace @{logonHours=$ct1a} } # CT1B: Tue/Thu 00:00-09:00 $ct1b = New-LogonArray Set-Hour $ct1b 2 0 9; Set-Hour $ct1b 4 0 9 Get-ADUser -Filter * -SearchBase "OU=CT1B,DC=group1,DC=local" | ForEach-Object { Set-ADUser $_ -Replace @{logonHours=$ct1b} } # CT2A: Mon-Wed 23:00-08:00 $ct2a = New-LogonArray for ($d=1; $d -le 3; $d++) { Set-Hour $ct2a $d 23 24; Set-Hour $ct2a $d 0 8 } Get-ADUser -Filter * -SearchBase "OU=CT2A,DC=group1,DC=local" | ForEach-Object { Set-ADUser $_ -Replace @{logonHours=$ct2a} } # CT2B: Thu-Sat 01:00-10:00 $ct2b = New-LogonArray for ($d=4; $d -le 6; $d++) { Set-Hour $ct2b $d 1 10 } Get-ADUser -Filter * -SearchBase "OU=CT2B,DC=group1,DC=local" | ForEach-Object { Set-ADUser $_ -Replace @{logonHours=$ct2b} } # CT3A: Mon 00:00-04:00 $ct3a = New-LogonArray; Set-Hour $ct3a 1 0 4 Get-ADUser -Filter * -SearchBase "OU=CT3A,DC=group1,DC=local" | ForEach-Object { Set-ADUser $_ -Replace @{logonHours=$ct3a} } # CT3B: Fri 05:00-09:00 $ct3b = New-LogonArray; Set-Hour $ct3b 5 5 9 Get-ADUser -Filter * -SearchBase "OU=CT3B,DC=group1,DC=local" | ForEach-Object { Set-ADUser $_ -Replace @{logonHours=$ct3b} } # CT4A: Sun/Sat 00:00-09:00 $ct4a = New-LogonArray; Set-Hour $ct4a 0 0 9; Set-Hour $ct4a 6 0 9 Get-ADUser -Filter * -SearchBase "OU=CT4A,DC=group1,DC=local" | ForEach-Object { Set-ADUser $_ -Replace @{logonHours=$ct4a} } # CT4B: Wed 02:00-07:00 $ct4b = New-LogonArray; Set-Hour $ct4b 3 2 7 Get-ADUser -Filter * -SearchBase "OU=CT4B,DC=group1,DC=local" | ForEach-Object { Set-ADUser $_ -Replace @{logonHours=$ct4b} } Write-Host "✅ ALL 8 Classes configured! Check ADUC to verify." -ForegroundColor Green
